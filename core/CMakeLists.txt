cmake_minimum_required(VERSION 3.15)
project(fidddle_core CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(fidddle_core STATIC
  src/spectral.cpp src/salience.cpp src/fitter.cpp src/tracker.cpp src/calibrator.cpp src/detector.cpp)
target_include_directories(fidddle_core PUBLIC include PRIVATE src)
target_compile_options(fidddle_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
set_target_properties(fidddle_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

# didddle: source separator. Depends on fidddle's public headers and (privately) its FFT; fidddle_core is unchanged.
add_library(didddle_core STATIC
  src/didddle/solver.cpp src/didddle/track_log.cpp src/didddle/separator.cpp src/didddle/splitter.cpp)
target_include_directories(didddle_core PUBLIC include PRIVATE src src/didddle)
target_link_libraries(didddle_core PUBLIC fidddle_core)
target_compile_options(didddle_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
set_target_properties(didddle_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(PROJECT_IS_TOP_LEVEL)
  add_executable(fidddle_cli cli/fidddle_cli.cpp)
  target_link_libraries(fidddle_cli fidddle_core)
  add_executable(fidddle_tests tests/test_core.cpp)
  target_include_directories(fidddle_tests PRIVATE src)
  target_link_libraries(fidddle_tests fidddle_core)
  enable_testing()
  add_test(NAME core COMMAND fidddle_tests)
  add_executable(didddle_tests tests/test_didddle.cpp)
  target_include_directories(didddle_tests PRIVATE src)
  target_link_libraries(didddle_tests didddle_core)
  add_test(NAME didddle COMMAND didddle_tests)
  add_executable(didddle_cli cli/didddle_cli.cpp)
  target_link_libraries(didddle_cli didddle_core)
endif()
