cmake_minimum_required(VERSION 3.15)
project(cellostop_core CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(cellostop_core STATIC
  src/spectral.cpp src/salience.cpp src/fitter.cpp src/tracker.cpp src/detector.cpp)
target_include_directories(cellostop_core PUBLIC include PRIVATE src)
target_compile_options(cellostop_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
set_target_properties(cellostop_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(PROJECT_IS_TOP_LEVEL)
  add_executable(cellostop_cli cli/cellostop_cli.cpp)
  target_link_libraries(cellostop_cli cellostop_core)
  add_executable(cellostop_tests tests/test_core.cpp)
  target_include_directories(cellostop_tests PRIVATE src)
  target_link_libraries(cellostop_tests cellostop_core)
  enable_testing()
  add_test(NAME core COMMAND cellostop_tests)
endif()
